# configuration file /etc/nginx/nginx.conf:
user www-data;
http {
    include /etc/nginx/sites-enabled/*;
}
# configuration file /etc/nginx/sites-enabled/app.conf:
server {
    listen 80;
    server_name app.example.test www.app.example.test;
    location /api/ {
        proxy_pass http://deploy:super-secret@127.0.0.1:5000;
        proxy_set_header Host $host;
    }
}
server {
    listen 443 ssl http2;
    server_name secure.example.test;
    ssl_certificate /etc/ssl/certs/secure.example.test.pem;
    ssl_certificate_key /etc/ssl/private/secure.example.test.key;
    location / {
        proxy_pass http://127.0.0.1:8080;
    }
}
